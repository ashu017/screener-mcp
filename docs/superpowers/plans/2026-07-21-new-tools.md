# New Tools (search / documents / compare) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add three anonymous Screener.in tools — `search_company`, `get_documents`, `compare_stocks` — plus a TTL cache on company-page fetches.

**Architecture:** Keep the existing two-file split: pure parsers + fetch helpers in `src/screener.ts`, tool wiring in `src/index.ts`. New parsers are pure functions over HTML/JSON so they're testable against the existing `test/tcs.fixture.html` fixture, mirroring `parseFundamentals`/`parsePeersFragment`. `compare_stocks` composes `fetchCompanyHtml` + `parseFundamentals`; a small module-local TTL cache in front of `fetchCompanyHtml` keeps fan-out from hammering Screener.

**Tech Stack:** TypeScript (ESM, NodeNext), `@modelcontextprotocol/sdk`, `node-html-parser`, `zod`, `vitest`.

## Global Constraints

- ESM with `.js` import specifiers (e.g. `from "./screener.js"`) — the project uses NodeNext.
- Anonymous only. No auth headers, cookies, or login. No new runtime dependencies.
- Reuse the existing `UA` constant and `fetch`; peers-style fragments need `X-Requested-With: XMLHttpRequest` (not needed for the endpoints here).
- Tools return `json(data)` on success and `err(message)` on failure — the existing helpers in `src/index.ts`.
- Parsers must be pure functions (no network) so fixture tests stay deterministic.
- Tests run with `npm test` (`vitest run`). Network tests carry an explicit timeout, matching the existing peers test.

```ts
/** Screener is server-rendered (Django). One GET returns the full page. Prefer
 * the consolidated view; fall back to standalone if consolidated 404s. */
async function fetchCompanyHtmlUncached(
  symbol: string,
): Promise<{ html: string; url: string }> {
  const sym = symbol.trim().toUpperCase();
  for (const view of ["consolidated/", ""]) {
    const path = `/company/${encodeURIComponent(sym)}/${view}`;
    const res = await fetch(`${BASE}${path}`, { headers: { "User-Agent": UA } });
    if (res.ok) return { html: await res.text(), url: `${BASE}${path}` };
  }
  throw new Error(`Screener: company '${sym}' not found (tried consolidated + standalone)`);
}

const COMPANY_TTL_MS = 5 * 60 * 1000;
const COMPANY_CACHE_MAX = 100;
const companyCache = new Map<string, { value: { html: string; url: string }; ts: number }>();

/** TTL-cached company page fetch. Keeps compare_stocks fan-out and repeated
 * tool calls from re-hitting Screener (README etiquette). Process-local only. */
export async function fetchCompanyHtml(
  symbol: string,
): Promise<{ html: string; url: string }> {
  const key = symbol.trim().toUpperCase();
  const hit = companyCache.get(key);
  if (hit && Date.now() - hit.ts < COMPANY_TTL_MS) return hit.value;
  const value = await fetchCompanyHtmlUncached(key);
  companyCache.set(key, { value, ts: Date.now() });
  if (companyCache.size > COMPANY_CACHE_MAX) {
    const oldest = companyCache.keys().next().value;
    if (oldest !== undefined) companyCache.delete(oldest);
  }
  return value;
}

/** Test-only: reset the company-page cache between tests. */
export function __clearCompanyCache(): void {
  companyCache.clear();
}
```

```ts
/** Parse Screener's /api/company/search/ JSON array into {symbol,name,id}.
 * Each entry looks like { id, name, url: "/company/TATASTEEL/consolidated/" }.
 * Entries without a parseable symbol are dropped. */
export function parseSearchResults(raw: unknown): SearchResult[] {
  if (!Array.isArray(raw)) return [];
  const out: SearchResult[] = [];
  for (const e of raw) {
    if (!e || typeof e !== "object") continue;
    const { id, name, url } = e as { id?: unknown; name?: unknown; url?: unknown };
    const symbol = typeof url === "string" ? url.match(/\/company\/([^/]+)\//)?.[1] : undefined;
    if (!symbol || typeof name !== "string" || typeof id !== "number") continue;
    out.push({ symbol, name, id });
  }
  return out;
}

/** Company name/partial -> matching listings. Anonymous JSON endpoint. */
export async function fetchSearch(query: string): Promise<SearchResult[]> {
  const path = `/api/company/search/?q=${encodeURIComponent(query)}`;
  const raw = await fetchHtml(path);
  return parseSearchResults(JSON.parse(raw));
}
```

```ts
/** Find the `ul.list-links` belonging to a Documents subsection by its <h3> text. */
function docListForHeader(section: HTMLElement, header: string): HTMLElement | null {
  const h3 = section.querySelectorAll("h3").find((h) => txt(h).toLowerCase() === header.toLowerCase());
  return h3?.parentNode?.querySelector("ul.list-links") ?? null;
}

/** Parse a plain link list (announcements / annual reports / credit ratings).
 * Title is the anchor text minus the inner meta div; date is that meta div's text. */
function parseDocLinks(list: HTMLElement | null): DocLink[] {
  if (!list) return [];
  const out: DocLink[] = [];
  for (const a of list.querySelectorAll("li a")) {
    const url = a.getAttribute("href");
    if (!url) continue;
    const meta = txt(a.querySelector("div"));
    const title = txt(a).replace(meta, "").trim();
    if (!title) continue;
    out.push({ title, date: meta || null, url });
  }
  return out;
}

export function parseDocuments(html: string): CompanyDocuments {
  const root = parse(html);
  const section = root.querySelector("#documents");
  if (!section) return { announcements: [], annualReports: [], creditRatings: [], concalls: [] };

  const announcements = parseDocLinks(docListForHeader(section, "Announcements"));
  const annualReports = parseDocLinks(docListForHeader(section, "Annual reports"));
  const creditRatings = parseDocLinks(docListForHeader(section, "Credit ratings"));

  const concalls: Concall[] = [];
  const concallList = docListForHeader(section, "Concalls");
  if (concallList) {
    for (const li of concallList.querySelectorAll("li")) {
      const date = txt(li.querySelector("div")) || null;
      for (const a of li.querySelectorAll("a.concall-link")) {
        const url = a.getAttribute("href");
        if (!url) continue;
        concalls.push({ date, kind: txt(a) || "link", url });
      }
    }
  }

  return { announcements, annualReports, creditRatings, concalls };
}
```

```ts
/** Align each company's ratio cards into one comparable row. When `metrics` is
 * given, keep only those ratio names (case-insensitive); otherwise keep all. */
export function compareFundamentals(items: Fundamentals[], metrics?: string[]): CompareRow[] {
  const want = metrics?.length ? metrics.map((m) => m.toLowerCase()) : null;
  return items.map((f) => {
    const ratios: Record<string, string> = {};
    for (const r of f.ratios) {
      if (want && !want.includes(r.name.toLowerCase())) continue;
      ratios[r.name] = r.value;
    }
    return { symbol: f.symbol, name: f.name, ratios };
  });
}
```

```ts
server.tool(
  "compare_stocks",
  "Compare an arbitrary list of Indian stocks side-by-side on their key ratios (from Screener.in). Unlike get_peers (auto sector peers), you choose the symbols. Optional 'metrics' filters which ratio names to include.",
  {
    symbols: z.array(z.string()).min(2).describe("NSE/BSE symbols to compare, e.g. ['TCS','INFY','WIPRO']"),
    metrics: z.array(z.string()).optional().describe("Optional ratio names to keep, e.g. ['Stock P/E','ROCE']"),

  async ({ symbols, metrics }) => {
    try {
      const settled = await Promise.all(
        symbols.map(async (s) => {
          try {
            const { html, url } = await fetchCompanyHtml(s);
            return parseFundamentals(s, html, url);
          } catch {
            return null;
          }
        }),
      );
      const resolved = settled.filter((f): f is NonNullable<typeof f> => f !== null);
      const missing = symbols.filter((_, i) => settled[i] === null).map((s) => s.toUpperCase());
      if (resolved.length === 0) return err(`No symbols resolved (missing: ${missing.join(", ")})`);
      return json({
        symbols: symbols.map((s) => s.toUpperCase()),
        metrics: metrics ?? null,
        rows: compareFundamentals(resolved, metrics),
        missing,
      });
    } catch (e) {
      return err(e instanceof Error ? e.message : String(e));
    }
  },
);
```
